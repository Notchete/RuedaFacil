CREATE DATABASE ruedafacil; 

CREATE TABLE CATEGORIA ( 
    nombre VARCHAR(50) PRIMARY KEY, 
    descripcion TEXT 
); 
 
CREATE TABLE CLIENTE ( 
    dni VARCHAR(15) PRIMARY KEY, 
    nombre VARCHAR(100) NOT NULL, 
    telefono VARCHAR(20), 
    correo VARCHAR(100), 
    direccion VARCHAR(200), 
    numCarnet VARCHAR(30) NOT NULL 
); 
 
CREATE TABLE EMPLEADO ( 
    idE INT AUTO_INCREMENT PRIMARY KEY, 
    nombre VARCHAR(100) NOT NULL, 
    puesto VARCHAR(50), 
    oficina VARCHAR(50), 
    turno VARCHAR(50), 
    aniosExp INT 
); 
 
CREATE TABLE VEHICULO ( 
    matricula VARCHAR(15) PRIMARY KEY, 
    marca VARCHAR(50) NOT NULL, 
    modelo VARCHAR(50) NOT NULL, 
    anioFabricacion INT, 
    combustible VARCHAR(50), 
    asientos INT, 
    precioPorDia DECIMAL(10,2) NOT NULL, 
    estadoV VARCHAR(50), 
    categoria VARCHAR(50), 
    FOREIGN KEY (categoria) REFERENCES CATEGORIA(nombre) ON UPDATE CASCADE 
ON DELETE SET NULL 
); 
 
CREATE TABLE ALQUILER ( 
    idA INT AUTO_INCREMENT PRIMARY KEY, 
    fechaInicio DATE NOT NULL, 
    fechaDevolucionPrevista DATE NOT NULL, 
    fechaDevolucion DATE, 
    precio DECIMAL(10,2), 
    estadoA VARCHAR(50), 
    dni VARCHAR(15), 
    idE INT, 
    matricula VARCHAR(15), 
    FOREIGN KEY (dni) REFERENCES CLIENTE(dni) ON UPDATE CASCADE ON DELETE 
CASCADE, 
    FOREIGN KEY (idE) REFERENCES EMPLEADO(idE) ON UPDATE CASCADE ON DELETE 
SET NULL, 
    FOREIGN KEY (matricula) REFERENCES VEHICULO(matricula) ON UPDATE CASCADE 
ON DELETE CASCADE 
);
